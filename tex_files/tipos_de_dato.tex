\section{Diferencias entre tipo}

\begin{frame}{Tipos de datos}
    A nivel práctico de manejo de R, podemos considerar que R solo cuenta con:
    \begin{itemize}
        \item double (númerico)
        \item character 
        \item bool
    \end{itemize}
    R si cuenta con datos tipo integer, string, entre otros. Pero para efectos prácticos, 
    solo se operan los anteriores
\end{frame}

\begin{frame}{Tipos de variables}
        \begin{center}
        \begin{tabular}{|c|p{11cm}|}
            \hline
            \textbf{Tipo de dato} & \textbf{Característica} \\ \hline
            Vector & Guarda los datos de un mismo tipo. Se le aplican la mayoría de funciones a nivel operativo. \\ \hline
            Factor & Para categorías y contruir datos categóricos \\ \hline
            DataFrame & Conjunto de datos, la manera estándar de operar la 'data' \\ \hline
            Table & Mayormente utilizada para frecuencias, dependiente de las categorías \\ \hline
            Lists & Guardar datos de varios tipos distintos \\ \hline
            Array & Vectores de tamaño especifico \\ \hline
            Matrix & Un herramienta de manejo de datos a nivel tensorial \\
            \hline
        \end{tabular}
    \end{center}
\end{frame}

\section{Comandos importantes}

\begin{frame}{Comandos que nunca vienen mal}
    \begin{center}
        \begin{tabular}{|c|p{11cm}|}
            \hline
            \textbf{Comando} & \textbf{Funcionalidad} \\ \hline
            SessionInfo() & Brinda los detalles completo de lo que se está utilizando \\ \hline
            ?\textit{función} & Muestra la documentación interna sobre \textit{función} \\ \hline
            ??\textit{función} & Muestra todas las funciones con nombres similares o dónde 
                                aparezca menionada \textit{función} \\ \hline
            typeof() & Descripe el tipo de dato o variable \\ \hline
            as.\textit{type}() & Modifica el tipo de variable al indicado según \textit{type} \\ \hline
            summary() & Da un resumen de las características de una varibale\\ \hline
        \end{tabular}
    \end{center}

\end{frame}

\begin{frame}{Comandos para dataframes}
    si se tiene un dataframe '\textit{df}', entonces:
    \begin{center}
        \begin{tabular}{|c|p{9cm}|}
            \hline
            \textbf{Comando} & \textbf{Funcionalidad} \\ \hline
            rbind(\textit{df},\textit{otro\_df}) & A \textit{df} le concatena \textit{otro\_df} \\ \hline
            duplicated(df) & Obtiene un vector booleano de qué renglones son duplicados \\ \hline
            order(df\$columna) & Obtiene un vector con el orden de las filas según una columna \\ \hline
            subset(data, condición) & Filtra de los datos que cumplean con la condición especificada \\ \hline
            tapply(v1,v2,\textit{función}) & Aplica la \textit{función} al v1 por cada v2. Especialmente útil cuando v1, 
                                            es un dato númerico y v2 una categoría \\
            \hline
        \end{tabular}
    \end{center}
\end{frame}

\section{Probabilidad}
\begin{frame}{Prefijo clave}
    \begin{center}
        \begin{tabular}{|c|p{12cm}|}
            \hline
            \textbf{Prefijo} & \textbf{Efecto} \\ \hline
            r & Genera una cantidad específica de datos aleatorios de la distribución especificada \\ \hline
            p & Obtiene el valor de la función de probabilidad acumulada en un punto dado de la distribución especificada \\ \hline
            d & Obtiene el valor de la función de densidad de probabilidad en un punto dado de la distribución especificada \\ \hline
            q & La inversa de la función de probabilidad acumulada. Indica el valor de la variable aleatoria con una cierta probabilidad acumulada \\\hline
        \end{tabular}
    \end{center}
\end{frame}
